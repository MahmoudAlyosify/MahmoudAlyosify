\cvsection{Research Experience \& Projects}

\cvproject{Verbosity-Aware Decoding for Token-Efficient Language Model Inference}
          {Master's Research Project (CISC 898), Queen's University \textperiodcentered\ Supervisor: Dr.\ Rahatara Ferdousi \textperiodcentered\ \href{https://github.com/MahmoudAlyosify/minimal-lm}{code}}
          {2026 -- Present}
\begin{cvitems}
  \item \textbf{Research question:} can generated-token count be reduced at inference time without retraining and without measurable loss of semantic content?
  \item \textbf{Method:} a \textbf{training-free LogitsProcessor} fusing stop-token, n-gram-repetition and semantic-redundancy signals under an \textbf{entropy-driven gate}; adds no KV-cache or attention overhead and requires no architecture change.
  \item \textbf{Design:} \textbf{pre-registered evaluation protocol} testing semantic preservation as a \textbf{paired non-inferiority hypothesis} (BERTScore, ROUGE, exact-match/F1, length-controlled LLM judge) --- the correct statistical framing for a "no worse than" claim, which conventional significance testing cannot supply.
  \item \textbf{Scope:} \textbf{three open-weight model families} (Mistral-7B-Instruct, Llama-3.1-8B-Instruct, Qwen2.5-3B-Instruct) across \textbf{four benchmarks} (MT-Bench, YapBench, XSum, SQuAD-v2), reporting token reduction, TTFT / inter-token latency and \textbf{GPU energy via NVML}.
  \item \textbf{Reproducibility:} deterministic runs with fixed seeds and versioned environments on pinned AWS GPU instances.
\end{cvitems}
