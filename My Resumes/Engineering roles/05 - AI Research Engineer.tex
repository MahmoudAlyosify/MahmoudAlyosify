\cvsection{Research Projects}

\cvproject{\href{https://github.com/MahmoudAlyosify/minimal-lm}{Verbosity-Aware Decoding for Token-Efficient Language Model Inference}}
          {MSc Research Project (CISC 898), Queen's University \textperiodcentered\ Supervisor: Dr.\ Rahatara Ferdousi}
          {2026 -- Present}
\begin{cvitems}
  \item \textbf{Contribution:} a \textbf{training-free LogitsProcessor} that reduces generated tokens at \textbf{decoding time} by fusing stop-token, n-gram-repetition and semantic-redundancy signals under an \textbf{entropy-driven gate} --- requiring no retraining, no architecture change and adding no KV-cache or attention overhead.
  \item \textbf{Generality:} validated across \textbf{three open-weight model families} (Mistral-7B-Instruct, Llama-3.1-8B-Instruct, Qwen2.5-3B-Instruct) and \textbf{four benchmarks} (MT-Bench, YapBench, XSum, SQuAD-v2) spanning open-ended chat, summarisation and extractive QA.
  \item \textbf{Rigour:} designed a \textbf{pre-registered protocol} that tests semantic preservation as a \textbf{paired non-inferiority hypothesis} (BERTScore, ROUGE, exact-match/F1, length-controlled LLM judge) rather than reporting a favourable mean --- alongside token reduction, TTFT / inter-token latency and \textbf{GPU energy} via NVML.
  \item \textbf{Reproducibility:} fully deterministic runs on pinned AWS GPU instances with fixed seeds and versioned environments.
\end{cvitems}
