\cvsection{Selected NLP Projects}

\cvproject{\href{https://github.com/MahmoudAlyosify/minimal-lm}{Verbosity-Aware Decoding for Token-Efficient Language Model Inference}}
          {MSc Research Project (CISC 898), Queen's University \textperiodcentered\ Supervisor: Dr.\ Rahatara Ferdousi}
          {2026 -- Present}
\begin{cvitems}
  \item Built a \textbf{training-free LogitsProcessor} operating directly on the decoding loop, combining \textbf{stop-token}, \textbf{n-gram-repetition} and \textbf{semantic-redundancy} signals under an \textbf{entropy-driven gate} to cut generated tokens without retraining or architecture change.
  \item Evaluated across three distinct NLP task types --- \textbf{open-ended dialogue (MT-Bench, YapBench)}, \textbf{abstractive summarisation (XSum)} and \textbf{extractive QA (SQuAD-v2)} --- on Mistral-7B, Llama-3.1-8B and Qwen2.5-3B.
  \item Designed a \textbf{pre-registered protocol} proving semantic preservation through paired \textbf{non-inferiority testing} using \textbf{BERTScore, ROUGE, exact-match/F1} and a \textbf{length-controlled LLM judge} --- controlling for the length bias that quietly inflates most generation metrics.
\end{cvitems}
