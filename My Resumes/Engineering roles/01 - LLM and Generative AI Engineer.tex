\cvsection{Selected LLM Research \& Engineering Projects}

\cvproject{\href{https://github.com/MahmoudAlyosify/minimal-lm}{Verbosity-Aware Decoding for Token-Efficient Language Model Inference}}
          {MSc Research Project (CISC 898), Queen's University \textperiodcentered\ Supervisor: Dr.\ Rahatara Ferdousi}
          {2026 -- Present}
\begin{cvitems}
  \item Designed a \textbf{training-free LogitsProcessor} that reduces generated tokens at \textbf{decoding time} by fusing stop-token, n-gram-repetition and semantic-redundancy signals under an \textbf{entropy-driven gate} --- no retraining, no architecture change, no KV-cache or attention overhead, and drop-in compatible with Hugging Face \texttt{generate}.
  \item Validated across \textbf{three open-weight model families} (Mistral-7B-Instruct, Llama-3.1-8B-Instruct, Qwen2.5-3B-Instruct) on \textbf{MT-Bench, YapBench, XSum} and \textbf{SQuAD-v2}, covering open-ended chat, summarisation and extractive QA.
  \item Built a \textbf{pre-registered evaluation protocol} that proves semantic preservation via paired \textbf{non-inferiority testing} (BERTScore, ROUGE, exact-match/F1, length-controlled LLM judge) alongside token reduction, \textbf{TTFT / inter-token latency} and \textbf{GPU energy} through NVML --- fully deterministic and reproducible on pinned AWS GPU instances.
\end{cvitems}
